#!name glob simple
mkdir -p d; touch d/a d/b d/ab; echo d/*; echo d/a*; echo d/?
%%%
#!name glob no match
echo nosuchdir/*
%%%
#!name glob bracket
touch xa xb xc; echo x[ab]; echo x[!a]; echo x[a-b]
%%%
#!name glob escaped star
touch 'lit*'; echo lit\*; echo 'lit*'; echo lit*
%%%
#!name glob dot files
touch .hidden vis; echo *; echo .*
%%%
#!name glob nested dirs
mkdir -p a/b/c; touch a/b/c/f; echo a/*/*/f; echo a/b/*
%%%
#!name glob trailing slash
mkdir -p g1 g2; touch gf; echo g*/
%%%
#!name glob with ctlesc pattern
touch 'a b'; echo a\ *; echo "a b"
%%%
#!name glob bracket unclosed
touch 'x['; echo 'x['*; echo x[
%%%
#!name glob only rbracket
echo ]
%%%
#!name glob question in dir part
mkdir -p q1; touch q1/z; echo ?1/z
%%%
#!name glob var expanded
p='*'; touch va vb; echo $p; echo "$p"
%%%
#!name glob noglob
set -f; touch fa fb; echo f*
