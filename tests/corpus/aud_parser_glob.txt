#!name glob quoted star mix

%%%
#!name glob quoted question
touch 'a?b' aXb; echo a"?"b; echo a'?'b; echo a\?b; echo a?b
%%%
#!name glob quoted bracket
touch 'a[b]c' aXc; echo 'a[b]c'; echo a[b]c; echo a\[b\]c; echo "a[b]c"
%%%
#!name glob partial quote
touch axb; echo "a"*b; echo a*"b"; echo "a*"b; echo a"*b"
%%%
#!name glob backslash before star
touch 'a\b' ab; echo a\\*; echo 'a\'*; echo "a\\"*
%%%
#!name glob var with meta
touch p1 p2; v='p*'; echo $v; echo "$v"; echo ${v}; echo "${v}"
%%%
#!name glob var quoted meta
touch 'p*'; v='p*'; echo "$v"
%%%
#!name glob esc in var
touch 'p*' pa; v='p\*'; echo $v; echo "$v"
%%%
#!name glob trailing backslash
touch 'x'; echo x\\ 2>&1 | head -1
%%%
#!name glob only backslash
echo \\ | od -An -c | head -1
%%%
#!name glob multiple backslash
echo \\\\ | od -An -c | head -1
%%%

%%%
#!name pattern trim bracket quoted

%%%
#!name pattern trim backslash content

#!name pattern trim mixed quotes

#!name pattern trim unquoted outer
