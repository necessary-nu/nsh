#!name fc -l with an empty pattern
#!mode=stdin
#!shargs -i
echo a
echo b
fc -l ''
%%%
#!name fc -l with a lone dash
#!mode=stdin
#!shargs -i
echo a
fc -l - 2>&1
%%%
#!name fc -l with a double dash
#!mode=stdin
#!shargs -i
echo a
fc -l -- 2>&1
%%%
#!name fc number before option stops parsing
#!mode=stdin
#!shargs -i
echo a
echo b
fc 1 -l 2>&1 | sed 's|/tmp/_sh[A-Za-z0-9]*|TMPFILE|g'
%%%
#!name fc -le combined
#!mode=stdin
#!shargs -i
echo a
fc -le 2>&1 | sed 's|/tmp/_sh[A-Za-z0-9]*|TMPFILE|g'

#!mode=stdin
#!shargs -i

fc -s = 2>&1

#!name HISTSIZE empty string means 100
#!mode=stdin

#!name HISTSIZE unset restores default
#!mode=stdin

echo d
fc -l 1 99 2>&1
%%%
#!name HISTSIZE with leading spaces
#!mode=stdin

%%%
#!name fc -s inside a compound command
#!mode=stdin
#!shargs -i
echo a
for i in 1
do
fc -l 1 99
done
%%%
#!name fc -l then a continuation line appends to the right event
#!mode=stdin
#!shargs -i
echo a
echo b
fc -l 1 1
for i in x
do
echo $i
done
fc -l 1 99
%%%
#!name fc pattern matching after fc moved the cursor
#!mode=stdin
#!shargs -i

echo a
fc -l 1 1
fc -l zmark 2>&1
%%%

#!mode=stdin
