#!name fc -l default range
#!mode=stdin
#!shargs -i
echo one
echo two
echo three
fc -l
%%%
#!name fc -l with explicit first
#!mode=stdin
#!shargs -i
echo a
echo b
echo c
echo d
fc -l 2
%%%
#!name fc -l with first and last
#!mode=stdin
#!shargs -i
echo a
echo b
echo c
echo d
fc -l 2 3
%%%
#!name fc -ln suppresses numbers
#!mode=stdin
#!shargs -i
echo a
echo b
fc -ln 1 2
%%%
#!name fc -lr reverses the range
#!mode=stdin
#!shargs -i
echo a
echo b
echo c
fc -lr 1 3
%%%
#!name fc -l with negative offset
#!mode=stdin
#!shargs -i
echo a
echo b
echo c
echo d
fc -l -2
%%%
#!name fc pattern search (H_PREV_STR)
#!mode=stdin
#!shargs -i
echo alpha
echo beta
true
fc -l echo
%%%
#!name fc -s re-executes the last command
#!mode=stdin
#!shargs -i
echo repeatme
fc -s
%%%
#!name fc -s with substitution
#!mode=stdin
#!shargs -i
echo original
fc -s original=replaced
%%%
#!name fc -s by event number
#!mode=stdin
#!shargs -i
echo first
echo second
fc -s 1
%%%
#!name fc numbering continues across many entries
#!mode=stdin
#!shargs -i
echo 1
echo 2
echo 3
echo 4
echo 5
fc -l 1 5
%%%
#!name fc on an unknown pattern is an error
#!mode=stdin
#!shargs -i
echo a
fc -l nosuchprefix
%%%
#!name fc with too many operands
#!mode=stdin
#!shargs -i
echo a
fc -l 1 2 3
%%%
#!name fc with an unknown option
#!mode=stdin
#!shargs -i
echo a
fc -Z
%%%
#!name multi-line command becomes one history event (H_APPEND)
#!mode=stdin
#!shargs -i
for i in a b
do
echo $i
done
fc -l
%%%
#!name HISTSIZE bounds the list
#!mode=stdin
#!shargs -i
HISTSIZE=3
echo a
echo b
echo c
echo d
echo e
fc -l 1 9
%%%
#!name fc -e with a non-existent editor
#!mode=stdin
#!shargs -i
echo a
fc -e /nonexistent/editor 1 2>&1 | sed 's|_sh[A-Za-z0-9]*|_shSUFFIX|'
%%%
#!name history is not recorded for a non-interactive shell
#!mode=stdin
echo a
fc -l
